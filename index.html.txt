      return new Intl.NumberFormat('id-ID', {
        style: 'currency',
        currency: 'IDR',
        maximumFractionDigits: 0

      if (Math.abs(num) >= 1000000) {
        return (num / 1000000).toFixed(1) + 'jt';
      }
      if (Math.abs(num) >= 1000) {
        return (num / 1000).toFixed(0) + 'rb';
      }
      return num.toString();

    const triggerHaptic = (pattern = 15) => {
      if (typeof window !== "undefined" && "navigator" in window && typeof navigator.vibrate === "function") {
        try {
          navigator.vibrate(pattern);
        } catch (e) {}
      }

    // --- Retro Audio Synthesizer ---
    class RetroAudio {
      constructor() {
        this.ctx = null;
        this.enabled = true;
      }

      getContext() {
        if (!this.ctx) {
          const AudioContextClass = window.AudioContext || window.webkitAudioContext;
          if (AudioContextClass) {
            this.ctx = new AudioContextClass();
          }
        }
        if (this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
        return this.ctx;
      }

      playClick() {
        triggerHaptic(15);
        if (!this.enabled) return;
        try {
          const ctx = this.getContext();
          if (!ctx) return;
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'square';
          osc.frequency.setValueAtTime(400, ctx.currentTime);
          osc.frequency.exponentialRampToValueAtTime(800, ctx.currentTime + 0.04);
          gain.gain.setValueAtTime(0.08, ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.04);
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start();
          osc.stop(ctx.currentTime + 0.04);
        } catch (e) {}
      }

      playSuccess() {
        triggerHaptic([20, 40, 20]);
        if (!this.enabled) return;
        try {
          const ctx = this.getContext();
          if (!ctx) return;
          const notes = [523.25, 659.25, 783.99, 1046.50];
          notes.forEach((freq, idx) => {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = 'triangle';
            osc.frequency.value = freq;
            const startTime = ctx.currentTime + idx * 0.06;
            gain.gain.setValueAtTime(0.1, startTime);
            gain.gain.exponentialRampToValueAtTime(0.001, startTime + 0.12);
            osc.connect(gain);
            gain.connect(ctx.destination);
            osc.start(startTime);
            osc.stop(startTime + 0.12);
          });
        } catch (e) {}
      }

      playCoin() {
        triggerHaptic([15, 30, 25]);
        if (!this.enabled) return;
        try {
          const ctx = this.getContext();
          if (!ctx) return;
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'square';
          osc.frequency.setValueAtTime(987.77, ctx.currentTime);
          osc.frequency.setValueAtTime(1318.51, ctx.currentTime + 0.08);
          gain.gain.setValueAtTime(0.12, ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.25);
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start();
          osc.stop(ctx.currentTime + 0.25);
        } catch (e) {}
      }

      playError() {
        triggerHaptic([40, 30, 40]);
        if (!this.enabled) return;
        try {
          const ctx = this.getContext();
          if (!ctx) return;
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(150, ctx.currentTime);
          osc.frequency.exponentialRampToValueAtTime(60, ctx.currentTime + 0.15);
          gain.gain.setValueAtTime(0.12, ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.15);
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start();
          osc.stop(ctx.currentTime + 0.15);
        } catch (e) {}
      }
    }

            </h3>

            <div className="space-y-3">
              {goals.map((g) => {

                    {!isCompleted && (
                      <button

                        className="w-full py-1.5 bg-[#83d4d9] text-[#003437] pixel-border font-bold text-xs button-active flex items-center justify-center gap-1"

                        <span className="material-symbols-outlined text-[14px]">payments</span>
                        <span>SETOR KE TARGET INI</span>
                      </button>

            </div>
          </section>

          {/* Modal Setor Tabungan */}
          {selectedGoalId && (
            <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60">
              <div className="bg-[#FFFDF7] pixel-border window-shadow w-full max-w-xs p-4 space-y-3">
                <h3 className="font-bold text-xs uppercase text-[#00696e]">SETOR SALDO KE TABUNGAN</h3>
                <p className="text-[10px] font-mono text-[#6e797a]">Saldo bebas tersedia: {formatRp(saldoBebas)}</p>
                <form onSubmit={handleDeposit} className="space-y-3">
                  <input
                    type="number"
                    placeholder="Nominal setor..."
                    value={depositAmount}
                    onChange={(e) => setDepositAmount(e.target.value)}
                    className="w-full p-2 bg-[#FFFDF7] pixel-border font-mono text-sm font-bold text-[#1b1b1c] outline-none"
                    required
                  />
                  <div className="flex justify-end gap-2">
                    <button
                      type="button"
                      onClick={() => setSelectedGoalId(null)}
                      className="px-3 py-1 bg-[#e5e2e1] text-[#1b1b1c] pixel-border text-xs font-bold button-active"
                    >
                      BATAL
                    </button>
                    <button
                      type="submit"
                      className="px-3 py-1 bg-[#00696e] text-white pixel-border text-xs font-bold button-active"
                    >
                      SETOR [ OK ]
                    </button>
                  </div>
                </form>
              </div>
            </div>
          )}
        </div>
      );
    };

    // --- COMPONENT: Config View ---
    const ConfigView = ({ soundEnabled, onToggleSound, transactions, goals, deposits, onImportData, onResetData }) => {

      const handleSelectChar = (char) => {
        soundEffects.playSuccess();
        setSelectedCharId(char.id);
        localStorage.setItem("selected_retro_char", char.id);
      };

      const handleExport = () => {
        soundEffects.playClick();
        const dataObj = { app: "FinancialManager", version: "1.0", transactions, goals, deposits };
        const blob = new Blob([JSON.stringify(dataObj, null, 2)], { type: "application/json" });
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");
        a.href = url;
        a.download = `backup_${new Date().toISOString().split("T")[0]}.json`;
        a.click();
        URL.revokeObjectURL(url);
        setStatusMessage("Export data JSON berhasil!");
      };

      return (
        <div className="space-y-4">
          <section className="bg-[#FFFDF7] pixel-border window-shadow p-4 space-y-3">
            <div className="h-7 bg-[#fe8b6d] pixel-border border-t-0 border-x-0 -m-4 mb-2 px-2 flex justify-between items-center text-[#75240e]">
              <span className="font-bold text-xs uppercase flex items-center gap-1">
                <span className="material-symbols-outlined text-[14px]">settings</span>
                SYSTEM CONFIG & MASCOT
              </span>
              <span className="text-[10px] bg-[#FFFDF7] px-1 font-mono pixel-border">v1.0</span>
            </div>

              <div className="p-2 bg-[#83d4d9] text-[#003437] text-xs font-mono pixel-border flex justify-between items-center">
                <span>{statusMessage}</span>
                <button onClick={() => setStatusMessage(null)}>✕</button>
              </div>

            {/* Active Mascot */}
            <div className="p-3 pixel-border flex items-start gap-3" style={{ backgroundColor: activeChar.bgColor, borderColor: activeChar.borderColor }}>
              <div className="w-12 h-12 pixel-border flex items-center justify-center text-white shrink-0" style={{ backgroundColor: activeChar.color }}>
                <span className="material-symbols-outlined text-[28px]">{activeChar.icon}</span>
              </div>
              <div className="space-y-1">
                <p className="font-bold text-xs uppercase" style={{ color: activeChar.textColor }}>{activeChar.name}</p>
                <p className="text-[10px] font-mono text-[#6e797a]">{activeChar.role}</p>
                <div className="p-1.5 bg-[#FFFDF7] pixel-border text-[11px] font-mono italic">"{activeChar.quote}"</div>
              </div>
            </div>

            {/* Mascot Grid */}
            <div className="grid grid-cols-2 gap-2">
              {RETRO_CHARACTERS.map((char) => (
                <button
                  key={char.id}
                  onClick={() => handleSelectChar(char)}
                  className={`p-2 pixel-border flex items-center gap-2 text-left ${char.id === selectedCharId ? 'bg-[#1b1b1c] text-white' : 'bg-[#f6f3f2] text-[#1b1b1c]'}`}
                >
                  <div className="w-7 h-7 pixel-border flex items-center justify-center text-white shrink-0" style={{ backgroundColor: char.color }}>
                    <span className="material-symbols-outlined text-[16px]">{char.icon}</span>

                  <p className="font-bold text-[10px] truncate">{char.name}</p>

          </section>

          <section className="bg-[#FFFDF7] pixel-border window-shadow p-3.5 flex justify-between items-center">

              <p className="font-bold text-xs text-[#1b1b1c]">Efek Suara 8-Bit & Getar</p>

            </div>
            <button
              onClick={() => { soundEffects.playClick(); onToggleSound(); }}
              className={`px-3 py-1 pixel-border font-bold text-xs button-active ${soundEnabled ? 'bg-[#00696e] text-white' : 'bg-[#e5e2e1] text-[#6e797a]'}`}
            >
              {soundEnabled ? 'ON' : 'OFF'}

              <span className="material-symbols-outlined text-[16px]">download</span>
              <span>EXPORT BACKUP JSON</span>

            <button
              onClick={() => { soundEffects.playClick(); setShowResetConfirm(true); }}
              className="w-full py-2 bg-[#ba1a1a] text-white pixel-border font-bold text-xs button-active flex items-center justify-center gap-1"

    // --- MAIN APP COMPONENT ---
    const App = () => {
      const STORAGE_KEY = 'fin_manager_8bit_os_data_v1';

      const [activeTab, setActiveTab] = useState('dashboard');
      const [transactions, setTransactions] = useState(() => {
        try {
          const saved = localStorage.getItem(STORAGE_KEY + '_tx');
          return saved ? JSON.parse(saved) : [
            { type: "in", amount: 2400000, category: "Gaji", date: "2026-08-01", note: "Gaji Bulan Agustus" },
            { type: "out", amount: 35000, category: "Makanan", date: "2026-08-02", note: "Makan Siang Nasi Padang" }
          ];
        } catch (e) { return []; }
      });

      const [goals, setGoals] = useState(() => {
        try {
          const saved = localStorage.getItem(STORAGE_KEY + '_goals');
          return saved ? JSON.parse(saved) : [
            { id: "g1", title: "Sepatu Sneaker Retro", targetAmount: 750000, currentAmount: 300000, status: "ongoing" }
          ];
        } catch (e) { return []; }
      });

      const [deposits, setDeposits] = useState(() => {
        try {
          const saved = localStorage.getItem(STORAGE_KEY + '_dep');
          return saved ? JSON.parse(saved) : [];
        } catch (e) { return []; }
      });

      const [soundEnabled, setSoundEnabled] = useState(() => {
        try {
          const saved = localStorage.getItem(STORAGE_KEY + '_sound');
          if (saved !== null) {
            const parsed = JSON.parse(saved);
            soundEffects.enabled = parsed;
            return parsed;
          }
        } catch (e) {}
        return true;
      });

      useEffect(() => { localStorage.setItem(STORAGE_KEY + '_tx', JSON.stringify(transactions)); }, [transactions]);
      useEffect(() => { localStorage.setItem(STORAGE_KEY + '_goals', JSON.stringify(goals)); }, [goals]);
      useEffect(() => { localStorage.setItem(STORAGE_KEY + '_dep', JSON.stringify(deposits)); }, [deposits]);
      useEffect(() => { localStorage.setItem(STORAGE_KEY + '_sound', JSON.stringify(soundEnabled)); }, [soundEnabled]);

      const toggleSound = () => {
        const nextVal = !soundEnabled;
        soundEffects.enabled = nextVal;
        setSoundEnabled(nextVal);
      };

      const totalIn = transactions.filter(t => t.type === 'in').reduce((a, b) => a + b.amount, 0);
      const totalOut = transactions.filter(t => t.type === 'out').reduce((a, b) => a + b.amount, 0);
      const totalSavings = goals.reduce((a, b) => a + b.currentAmount, 0);
      const saldoBebas = totalIn - totalOut - totalSavings;

      const handleAddTransaction = (t) => setTransactions([t, ...transactions]);
      const handleDeleteTransaction = (idx) => setTransactions(transactions.filter((_, i) => i !== idx));

      const handleAddGoal = (g) => setGoals([...goals, g]);
      const handleDepositGoal = (goalId, amount) => {
        setGoals(goals.map(g => g.id === goalId ? { ...g, currentAmount: g.currentAmount + amount } : g));
        setDeposits([{ goalId, amount, date: new Date().toISOString().split('T')[0] }, ...deposits]);
      };

      const handleResetData = () => {
        setTransactions([]);
        setGoals([]);
        setDeposits([]);
        localStorage.clear();
      };

      return (
        <div className="flex-1 flex flex-col h-full bg-[#f6f3f2]">
          {/* Header OS Bar */}
          <header className="h-10 bg-[#1b1b1c] text-white pixel-border border-x-0 border-t-0 px-3 flex items-center justify-between select-none">
            <div className="flex items-center gap-2">
              <span className="material-symbols-outlined text-[18px] text-[#83d4d9]">videogame_asset</span>
              <span className="font-bold text-xs tracking-wider text-[#83d4d9]">FINANCIAL MANAGER OS</span>
            </div>
            <span className="text-[10px] font-mono bg-[#3e4949] px-1.5 py-0.5 text-white pixel-border">8-BIT</span>
          </header>

          {/* Main Content Area */}
          <main className="flex-1 p-3 overflow-y-auto custom-scrollbar pb-20">
            {activeTab === 'dashboard' && (
              <DashboardView
                saldoBebas={saldoBebas}
                totalIn={totalIn}
                totalOut={totalOut}
                totalSavings={totalSavings}
                onNavigate={setActiveTab}
              />
            )}
            {activeTab === 'transaksi' && (
              <TransaksiView
                transactions={transactions}
                onAddTransaction={handleAddTransaction}
                onDeleteTransaction={handleDeleteTransaction}
              />
            )}
            {activeTab === 'tabungan' && (
              <TabunganView
                goals={goals}
                deposits={deposits}
                saldoBebas={saldoBebas}
                onAddGoal={handleAddGoal}
                onDepositGoal={handleDepositGoal}
              />
            )}
            {activeTab === 'config' && (
              <ConfigView
                soundEnabled={soundEnabled}
                onToggleSound={toggleSound}
                transactions={transactions}
                goals={goals}
                deposits={deposits}
                onImportData={(data) => {
                  setTransactions(data.transactions || []);
                  setGoals(data.goals || []);
                  setDeposits(data.deposits || []);
                }}
                onResetData={handleResetData}
              />
            )}
          </main>

          {/* Bottom Dock Navigation Bar */}
          <nav className="fixed bottom-0 left-0 right-0 max-w-md mx-auto bg-[#1b1b1c] pixel-border border-x-0 border-b-0 p-1 grid grid-cols-4 gap-1 z-40">
            {[
              { id: 'dashboard', label: 'DASHBOARD', icon: 'dashboard' },
              { id: 'transaksi', label: 'TRANSAKSI', icon: 'receipt_long' },
              { id: 'tabungan', label: 'TABUNGAN', icon: 'savings' },
              { id: 'config', label: 'CONFIG', icon: 'settings' }
            ].map((tab) => (
              <button
                key={tab.id}
                onClick={() => { soundEffects.playClick(); setActiveTab(tab.id); }}
                className={`py-2 pixel-border flex flex-col items-center justify-center transition-all ${
                  activeTab === tab.id
                    ? 'bg-[#00696e] text-white'
                    : 'bg-[#3e4949] text-[#e5e2e1] hover:bg-[#50a2a7] hover:text-white'
                }`}
              >
                <span className="material-symbols-outlined text-[18px]">{tab.icon}</span>
                <span className="text-[9px] font-bold tracking-tight">{tab.label}</span>
              </button>
            ))}
          </nav>
        </div>
      );
    };

    // Render App
    const rootElement = document.getElementById('root');
    const root = ReactDOM.createRoot(rootElement);
    root.render(<App />);
  </script>
